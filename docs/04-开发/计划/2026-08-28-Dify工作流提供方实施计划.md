# Dify 工作流模型提供方实施计划

> 状态：已暂停。该计划针对纯 Workflow App 的 `/workflows/run`，与当前真实 `chat` 项目的 Chatflow `/chat-messages` 不一致。当前实施必须以[最小化实时语音全链路实施方案](../../02-架构/最小化实时语音全链路实施方案.md)为准，并在开始编码前生成新的 Chatflow 实施计划。

> **执行要求：** 使用 `superpowers:subagent-driven-development`（推荐）或 `superpowers:executing-plans` 按任务逐项实施；使用复选框（`- [ ]`）记录进度。

**目标：** 在 `xiaozhi-all` 的模型路由器中实现无状态、真流式、可取消、可降级的 Dify 工作流大语言模型提供方，并通过契约、集成、延迟和 ESP32 真机验收。

**架构：** Dify 适配器内置于 `services/model-router`，通过 `httpx.AsyncClient.stream()` 调用已经部署好的 Dify `/v1/workflows/run`。会话服务每轮传入角色、短期上下文和长期记忆；适配器将服务端事件流 `text_chunk` 归一化为统一大语言模型文本增量，并在用户打断时关闭流和调用工作流停止接口。Dify 不保存会话记忆，本项目不部署或扩缩容 Dify。

**技术栈：** Python 3.12、asyncio、httpx、Pydantic 2、pydantic-settings、pytest、pytest-asyncio、respx、OpenTelemetry 接口、Prometheus 客户端。

## 全局约束

- Dify 是已经部署好的外部服务；不得增加 Dify 部署、扩缩容、数据库、队列、执行进程或模型安装代码。
- Dify 工作流每轮无状态执行，不发送或保存 `conversation_id`。
- `xiaozhi-all` 是会话、轮次、生成任务、角色、短期历史、长期记忆和实际听到文本的唯一事实源。
- 仅使用 `response_mode=streaming`；禁止先等待完整 HTTP 响应再读取服务端事件流。
- 首个文本增量前最大网络重试次数为 1；首个文本增量后禁止重试和拼接备用回复。
- 用户打断必须先完成本地生成任务、语音合成和播放取消；Dify 停止接口只能作为向下游传播的尽力操作。
- 接口密钥只从服务端密钥配置注入，不得进入仓库、设备、网页端、移动端、日志、链路追踪或测试夹具。
- 大语言模型首字延迟目标 P95 不高于 450 毫秒；全链路首音 P95 不高于 1.2 秒；打断停止播放 P95 不高于 250 毫秒。
- 新增行为执行测试驱动开发：失败测试、最小实现、通过、相关回归。

```python
# services/model-router/src/model_router/contracts.py
from __future__ import annotations

from abc import ABC, abstractmethod
from typing import Any, AsyncIterator, Literal

from pydantic import BaseModel, Field


class LLMRequest(BaseModel):
    session_id: str
    turn_id: str
    generation_id: str
    user_id: str
    device_id: str | None = None
    user_text: str = Field(min_length=1)
    role_profile: dict[str, Any]
    short_history: list[dict[str, Any]] = Field(default_factory=list)
    long_memories: list[dict[str, Any]] = Field(default_factory=list)
    scene: Literal["companion_chat", "knowledge_qa", "tool_task"] = "companion_chat"
    locale: str = "zh-CN"
    response_style: dict[str, Any] = Field(default_factory=dict)


class LLMEvent(BaseModel):
    session_id: str
    turn_id: str
    generation_id: str
    provider: str


class LLMStarted(LLMEvent):
    type: Literal["llm.started"] = "llm.started"


class LLMTextDelta(LLMEvent):
    type: Literal["llm.text.delta"] = "llm.text.delta"
    sequence: int = Field(ge=1)
    text: str = Field(min_length=1)

    @classmethod
    def from_request(
        cls, request: LLMRequest, provider: str, sequence: int, text: str
    ) -> "LLMTextDelta":
        return cls(
            session_id=request.session_id,
            turn_id=request.turn_id,
            generation_id=request.generation_id,
            provider=provider,
            sequence=sequence,
            text=text,
        )


class LLMCompleted(LLMEvent):
    type: Literal["llm.completed"] = "llm.completed"
    reply_text: str
    finish_reason: str = "stop"
    usage: dict[str, int] = Field(default_factory=dict)
    metadata: dict[str, Any] = Field(default_factory=dict)


class LLMFailed(LLMEvent):
    type: Literal["llm.failed"] = "llm.failed"
    code: str
    retryable: bool
    delta_emitted: bool


class LLMCancelled(LLMEvent):
    type: Literal["llm.cancelled"] = "llm.cancelled"


LLMStreamEvent = LLMStarted | LLMTextDelta | LLMCompleted | LLMFailed | LLMCancelled


class LLMProvider(ABC):
    @abstractmethod
    async def stream(
        self, request: LLMRequest, cancel_event: "asyncio.Event"
    ) -> AsyncIterator[LLMStreamEvent]:
        raise NotImplementedError
```

```python
# services/model-router/src/model_router/providers/dify_workflow/event_parser.py
from __future__ import annotations

import json
from dataclasses import dataclass
from typing import Any


class DifyProtocolError(ValueError):
    pass


@dataclass(frozen=True, slots=True)
class DifyEvent:
    event: str
    task_id: str | None
    workflow_run_id: str | None
    data: dict[str, Any]


def parse_sse_line(line: str) -> DifyEvent | None:
    if not line or not line.startswith("data: "):
        return None
    try:
        payload = json.loads(line[6:])
    except json.JSONDecodeError as exc:
        raise DifyProtocolError("invalid Dify SSE JSON") from exc
    event_name = payload.get("event")
    if not isinstance(event_name, str) or not event_name:
        raise DifyProtocolError("missing Dify event name")
    data = payload.get("data") or {}
    if not isinstance(data, dict):
        raise DifyProtocolError("Dify event data must be an object")
    return DifyEvent(
        event=event_name,
        task_id=payload.get("task_id"),
        workflow_run_id=payload.get("workflow_run_id"),
        data=data,
    )
```

```python
# services/model-router/src/model_router/providers/dify_workflow/client.py
from __future__ import annotations

from typing import AsyncIterator

import httpx

from .config import DifyWorkflowSettings
from .event_parser import DifyEvent, parse_sse_line


class DifyWorkflowClient:
    def __init__(self, settings: DifyWorkflowSettings, http: httpx.AsyncClient) -> None:
        self.settings = settings
        self.http = http

    @property
    def headers(self) -> dict[str, str]:
        return {
            "Authorization": f"Bearer {self.settings.api_key.get_secret_value()}",
            "Accept": "text/event-stream",
            "Content-Type": "application/json",
        }

    async def stream(self, payload: dict[str, object]) -> AsyncIterator[DifyEvent]:
        url = f"{str(self.settings.base_url).rstrip('/')}/workflows/run"
        timeout = httpx.Timeout(
            connect=self.settings.connect_timeout_ms / 1000,
            read=self.settings.read_timeout_ms / 1000,
            write=2.0,
            pool=1.0,
        )
        async with self.http.stream(
            "POST", url, headers=self.headers, json=payload, timeout=timeout
        ) as response:
            response.raise_for_status()
            async for line in response.aiter_lines():
                event = parse_sse_line(line)
                if event is not None:
                    yield event

    async def stop(self, task_id: str, user: str) -> None:
        url = f"{str(self.settings.base_url).rstrip('/')}/workflows/tasks/{task_id}/stop"
        await self.http.post(
            url,
            headers=self.headers,
            json={"user": user},
            timeout=self.settings.stop_timeout_ms / 1000,
        )
```

```python
# services/model-router/src/model_router/providers/dify_workflow/provider.py
from __future__ import annotations

import asyncio
from typing import AsyncIterator

from model_router.contracts import (
    LLMCancelled, LLMCompleted, LLMFailed, LLMRequest,
    LLMStarted, LLMStreamEvent, LLMTextDelta,
)

from .client import DifyWorkflowClient
from .config import DifyWorkflowSettings
from .input_mapper import map_dify_request


class DifyWorkflowProvider:
    def __init__(self, settings: DifyWorkflowSettings, client: DifyWorkflowClient) -> None:
        self.settings = settings
        self.client = client

    def _base(self, request: LLMRequest) -> dict[str, str]:
        return {
            "session_id": request.session_id,
            "turn_id": request.turn_id,
            "generation_id": request.generation_id,
            "provider": self.settings.provider_id,
        }

    async def stream(
        self, request: LLMRequest, cancel_event: asyncio.Event
    ) -> AsyncIterator[LLMStreamEvent]:
        sequence = 0
        parts: list[str] = []
        task_id: str | None = None
        emitted_started = False
        async for event in self.client.stream(map_dify_request(request)):
            if cancel_event.is_set():
                if task_id:
                    try:
                        await self.client.stop(task_id, request.user_id)
                    except Exception:
                        pass
                yield LLMCancelled(**self._base(request))
                return
            task_id = event.task_id or task_id
            if event.event == "workflow_started" and not emitted_started:
                emitted_started = True
                yield LLMStarted(**self._base(request))
            elif event.event == "text_chunk":
                text = event.data.get("text")
                if isinstance(text, str) and text:
                    sequence += 1
                    parts.append(text)
                    yield LLMTextDelta.from_request(
                        request, self.settings.provider_id, sequence, text
                    )
            elif event.event == "workflow_finished":
                status = event.data.get("status")
                if status == "succeeded":
                    yield LLMCompleted(
                        **self._base(request), reply_text="".join(parts),
                        usage={}, metadata=event.data.get("outputs") or {},
                    )
                else:
                    yield LLMFailed(
                        **self._base(request), code="DIFY_WORKFLOW_FAILED",
                        retryable=False, delta_emitted=bool(parts),
                    )
                return
            elif event.event in {"workflow_paused", "human_input_required"}:
                yield LLMFailed(
                    **self._base(request), code="DIFY_UNSUPPORTED_PAUSE",
                    retryable=False, delta_emitted=bool(parts),
                )
                return
            elif event.event == "error":
                yield LLMFailed(
                    **self._base(request), code="DIFY_WORKFLOW_FAILED",
                    retryable=not parts, delta_emitted=bool(parts),
                )
                return
```

在最终实现中，用 `try/finally` 保证外部取消异常也会关闭流；Stop API 失败只记指标，不重新激活 Generation。

```python
# services/model-router/src/model_router/router.py
from __future__ import annotations

import asyncio
import uuid
from typing import AsyncIterator, Sequence

from model_router.contracts import LLMFailed, LLMProvider, LLMRequest, LLMStreamEvent


class LLMRouter:
    def __init__(self, providers: Sequence[LLMProvider]) -> None:
        if not providers:
            raise ValueError("at least one LLM provider is required")
        self.providers = providers

    async def stream(
        self, request: LLMRequest, cancel_event: asyncio.Event
    ) -> AsyncIterator[LLMStreamEvent]:
        current = request
        for index, provider in enumerate(self.providers):
            can_fallback = False
            async for event in provider.stream(current, cancel_event):
                yield event
                if isinstance(event, LLMFailed):
                    can_fallback = event.retryable and not event.delta_emitted
            if cancel_event.is_set() or not can_fallback or index == len(self.providers) - 1:
                return
            current = current.model_copy(
                update={"generation_id": f"g_{uuid.uuid4().hex}"}
            )
```

```python
import statistics
import time

import pytest


pytestmark = pytest.mark.skipif(
    os.getenv("RUN_DIFY_BENCHMARK") != "1",
    reason="set RUN_DIFY_BENCHMARK=1 for deployed Dify validation",
)


def percentile(values: list[float], fraction: float) -> float:
    ordered = sorted(values)
    index = min(len(ordered) - 1, int(len(ordered) * fraction))
    return ordered[index]


@pytest.mark.asyncio
async def test_dify_ttft_100_turns(dify_provider, benchmark_requests) -> None:
    assert len(benchmark_requests) == 100
    ttft_ms: list[float] = []
    for request in benchmark_requests:
        started = time.perf_counter()
        async for event in dify_provider.stream(request):
            if event.type == "llm.text.delta":
                ttft_ms.append((time.perf_counter() - started) * 1000)
                break
    assert len(ttft_ms) == 100
    assert percentile(ttft_ms, 0.95) <= 450
    print({"p50_ms": statistics.median(ttft_ms), "p95_ms": percentile(ttft_ms, 0.95)})
```
